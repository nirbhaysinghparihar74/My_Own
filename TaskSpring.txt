package SpringCore.Assingment;

import java.util.Objects;

public class Assingment3 {

	private int rollnumber;
	private String StudentName;
	private String course;
	

	public void setRollnumber(int rollnumber) {
		this.rollnumber = rollnumber;
	}
	
	public void setStudentName(String studentName) {
		StudentName = studentName;
	}
	
	public void setCourse(String course) {
		this.course = course;
		
	}
	
	public int getRollnumber() {
		return rollnumber;
	}

	public String getStudentName() {
		return StudentName;
	}

	public String getCourse() {
		return course;
	}
	
	@Override
	public String toString() {
		return "Assingment3 [rollnumber=" + rollnumber + ", StudentName=" + StudentName + ", course=" + course + "]";
	}



	@Override
	public int hashCode() {
		return Objects.hash(StudentName, course, rollnumber);
	}


	@Override
	public boolean equals(Object obj) {
		if (this == obj)
			return true;
		if (obj == null)
			return false;
		if (getClass() != obj.getClass())
			return false;
		Assingment3 other = (Assingment3) obj;
		return Objects.equals(StudentName, other.StudentName) && Objects.equals(course, other.course)
				&& rollnumber == other.rollnumber;
	}
}
